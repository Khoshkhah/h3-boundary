# Interactive Demo

The five stages of boundary tracing and buffering, on live maps. **Demo cell:** `872830828ffffff` (resolution 7, San Francisco), traced at resolution 10.

## 1. The cell itself

The plain hexagon H3 draws for the cell — `cell_boundary_to_geojson()`.

<div id="map1" class="map-container"></div>

## 2. Its boundary cells

`children_on_boundary_faces()` — the resolution-10 descendants on its edge. Note how they straddle the hexagon rather than staying inside it.

<div id="map2" class="map-container"></div>

## 3. The merged outline

`cell_boundary_from_children()` — those cells fused into one polygon: the cell's true footprint.

<div id="map3" class="map-container"></div>

## 4. Buffered, accurate

`get_buffered_boundary_polygon()` — the outline grown by a safety margin, guaranteed to contain every descendant.

<div id="map4" class="map-container"></div>

## 5. Buffered, fast

The same guarantee via a convex hull (`use_convex_hull=True`) — simpler and ~20× faster, at the cost of extra area. The accurate polygon is ghosted for comparison.

<div id="map5" class="map-container"></div>

<script>
  var tiles = 'https://{s}.basemaps.cartocdn.com/light_all/{z}/{x}/{y}{r}.png';
  var fallbackCenter = [37.773, -122.418];

  // Value from demo_data.js if it loaded, else null — guarded per variable
  // so one missing entry can't blank every map.
  function data(name) {
    return typeof window[name] !== 'undefined' ? window[name] : null;
  }

  // Creates a map, adds every non-null [geojson, style] pair, and fits the
  // view to the data so any cell size stays on screen.
  function makeMap(id, layers) {
    var map = L.map(id);
    L.tileLayer(tiles).addTo(map);
    var group = L.featureGroup();
    layers.forEach(function (l) {
      if (l[0]) L.geoJSON(l[0], { style: l[1] }).addTo(group);
    });
    if (group.getLayers().length) {
      group.addTo(map);
      map.fitBounds(group.getBounds(), { padding: [10, 10] });
    } else {
      map.setView(fallbackCenter, 13);
    }
    return map;
  }

  var cellOutline = { color: '#238636', weight: 1, fillOpacity: 0, dashArray: '4' };

  makeMap('map1', [
    [data('cellBoundary'), { color: '#238636', weight: 4, fillOpacity: 0.1 }]]);
  makeMap('map2', [
    [data('cellBoundary'), cellOutline],
    [data('boundaryChildren'), { color: '#f0883e', weight: 1, fillOpacity: 0.4 }]]);
  makeMap('map3', [
    [data('cellBoundary'), cellOutline],
    [data('mergedBoundary'), { color: '#1f6feb', weight: 3, fillOpacity: 0.3 }]]);
  makeMap('map4', [
    [data('cellBoundary'), cellOutline],
    [data('bufferedAccurate'), { color: '#f85149', weight: 3, fillOpacity: 0.3 }]]);
  makeMap('map5', [
    [data('bufferedAccurate'), { color: '#f85149', weight: 1, fillOpacity: 0.1, dashArray: '4' }],
    [data('bufferedFast'), { color: '#a371f7', weight: 3, fillOpacity: 0.3 }]]);
</script>

The data behind these maps is generated by [`docs/generate_demo_data.py`](https://github.com/Khoshkhah/h3-boundary/blob/master/docs/generate_demo_data.py); the same workflow runs in [`notebook/demo_generation.ipynb`](https://github.com/Khoshkhah/h3-boundary/blob/master/notebook/demo_generation.ipynb).
